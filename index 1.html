""" # ============================================================ # HOME PAGE # ============================================================ @app.route("/") def home(): return render_template_string(HTML) # ============================================================ # AI ANALYSIS # ============================================================ @app.route("/analyze", methods=["POST"]) def analyze(): data = request.get_json() name = data.get("name", "").strip() location = data.get("location", "").strip() complaint = data.get("complaint", "").strip() if not complaint: return jsonify({ "error": "Complaint cannot be empty." }), 400 # -------------------------------------------------------- # DEMO FALLBACK # -------------------------------------------------------- if client is None: return jsonify({ "category": "Street Lighting", "department": "Electricity Department", "priority": "Medium", "confidence": 91, "reason": "The complaint describes a non-working street light, so it should be handled by the department responsible for electricity and street lighting.", "summary": complaint, "recommended_action": "Inspect the reported street light and repair or replace the faulty lighting equipment." }) # ======================================================== # LLM PROMPT # ======================================================== prompt = f""" You are an intelligent government Citizen Complaint Categorization and Routing System. Your job is to analyze a citizen complaint and classify it into the correct government service category and department. IMPORTANT: Do NOT ask the citizen to choose a department. The AI must determine the department automatically. AVAILABLE CATEGORIES: 1. Water Supply 2. Electricity 3. Street Lighting 4. Roads & Infrastructure 5. Waste Management 6. Drainage 7. Education 8. Public Transport 9. Government Documents 10. Public Health DEPARTMENT MAPPING: Water Supply -> Water Supply Department Electricity -> Electricity Department Street Lighting -> Electricity Department Roads & Infrastructure -> Public Works Department Waste Management -> Sanitation Department Drainage -> Municipal Department Education -> Education Department Public Transport -> Transport Department Government Documents -> Revenue / E-Governance Department Public Health -> Health Department PRIORITY RULES: High: - immediate public danger - major accident risk - severe infrastructure failure - serious health/public safety issue - large number of people affected Medium: - important problem - repeated service failure - moderate public inconvenience Low: - minor issue - general information - small inconvenience CITIZEN: Name: {name} Location: {location} COMPLAINT: {complaint} Return ONLY valid JSON. Do not use Markdown. Use exactly this structure: {{ "category": "one category from the list", "department": "correct government department", "priority": "High, Medium, or Low", "confidence": 0, "reason": "short explanation", "summary": "short professional summary", "recommended_action": "what the department should do" }} The confidence must be an integer from 0 to 100. """ try: response = client.responses.create( model="gpt-5.6-luna", input=prompt ) text = response.output_text.strip() # Remove accidental Markdown JSON fences if text.startswith("```"): text = text.replace( "```json", "" ).replace( "```", "" ).strip() result = json.loads(text) # ---------------------------------------------------- # BASIC VALIDATION # ---------------------------------------------------- required_fields = [ "category", "department", "priority", "confidence", "reason", "summary", "recommended_action" ] for field in required_fields: if field not in result: raise ValueError( f"Missing field: {field}" ) # Keep confidence within 0-100 result["confidence"] = max( 0, min( 100, int(result["confidence"]) ) ) return jsonify(result) except json.JSONDecodeError: return jsonify({ "error": "The AI returned an invalid response. Please try again." }), 500 except Exception as e: return jsonify({ "error": "AI service error: " + str(e) }), 500 # ============================================================ # RUN APPLICATION # ============================================================ if __name__ == "__main__": app.run( debug=True, host="127.0.0.1", port=5000 )
